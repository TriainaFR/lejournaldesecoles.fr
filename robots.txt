# Content Signals — déclaration de nos préférences d'usage du contenu.
# Le Journal des Écoles VEUT être lu, indexé et cité par les moteurs de
# réponse : les trois signaux sont donc positifs, à l'inverse du réglage
# restrictif appliqué par défaut par certains CDN, qui interdit l'usage du
# contenu pour l'entraînement et neutraliserait notre stratégie GEO.
#   search    : indexation par les moteurs de recherche
#   ai-input  : utilisation comme source dans une réponse générée (RAG)
#   ai-train  : utilisation pour l'entraînement de modèles
# Reprise autorisée à condition de citer la source et de lier l'article.
# ————————————————————————————————————————————————

User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=yes
Allow: /

# Recherche interne : pages sans valeur pour un moteur, déjà en noindex.
# On bloque aussi leur exploration pour ne pas gaspiller le budget de crawl
# d'un site jeune sur des URL qui ne seront jamais indexées.
Disallow: /recherche/

# ————————————————————————————————————————————————
# Moteurs de réponse et crawlers IA — autorisés explicitement.
# Sans cela, aucune citation possible dans ChatGPT, Perplexity,
# Claude, Gemini ou les AI Overviews de Google.
# ————————————————————————————————————————————————
